cmake_minimum_required(VERSION 3.16)
project(haeunos VERSION 0.1.0 LANGUAGES C)

set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

add_library(haeun_vm STATIC
  src/vm.c src/board_mmio.c src/board_extra.c src/board_net.c src/trap.c src/mmu.c src/tlb.c src/pt.c
  src/register_file.c src/cpu_state.c src/memory_interface.c
  src/instruction_decoder.c src/instruction_executor.c src/instruction_trace.c
  src/sec_core.c src/sec_dispatch.c src/sec_ecall.c src/sha256.c src/sec_boot.c
  src/hxf.c src/hfw.c src/bootinfo.c src/disk_persist.c)
target_include_directories(haeun_vm PUBLIC include)
if(MSVC)
  target_compile_options(haeun_vm PRIVATE /W4 /WX)
else()
  target_compile_options(haeun_vm PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

add_library(haeun_hfs STATIC src/hfs_io.c src/hfs_mkfs.c src/hfs_ops.c src/hfs_fsck.c
  src/proc_hfs.c)
target_include_directories(haeun_hfs PUBLIC include)
if(MSVC)
  target_compile_options(haeun_hfs PRIVATE /W4 /WX)
else()
  target_compile_options(haeun_hfs PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

add_library(haeun_gfx STATIC src/gfx_fb.c src/gfx_pixel.c src/gfx_font.c
  src/gfx_wm.c src/gfx_comp.c src/gfx_ui.c src/gfx_desk.c src/gfx_ecall.c)

target_include_directories(haeun_gfx PUBLIC include)
target_link_libraries(haeun_gfx PUBLIC haeun_vm)

if(MSVC)
  target_compile_options(haeun_gfx PRIVATE /W4 /WX)
else()
  target_compile_options(haeun_gfx PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

add_library(haeun_net STATIC src/net_eth.c src/net_arp.c src/net_ipv4.c
  src/net_udp.c src/net_tcp.c src/net_dns.c src/net_sock.c src/net_drv.c
  src/net_sock_ecall.c)
target_include_directories(haeun_net PUBLIC include)
target_link_libraries(haeun_net PUBLIC haeun_vm)
if(MSVC)
  target_compile_options(haeun_net PRIVATE /W4 /WX)
else()
  target_compile_options(haeun_net PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

# Security sources live in haeun_vm (ECALL gate). Alias for existing targets.
add_library(haeun_sec INTERFACE)
target_link_libraries(haeun_sec INTERFACE haeun_vm)

add_executable(haeun-vm src/main.c)
target_link_libraries(haeun-vm PRIVATE haeun_vm)

add_executable(haeun-mkfs tools/haeun_mkfs.c)
target_link_libraries(haeun-mkfs PRIVATE haeun_hfs)
add_executable(haeun-fsck tools/haeun_fsck.c)
target_link_libraries(haeun-fsck PRIVATE haeun_hfs)
